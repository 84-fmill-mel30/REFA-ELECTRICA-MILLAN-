const TIPO_FILL  = { sensor:"#051525",inyector:"#1a0530",solenoide:"#251505",relay:"#051a10",bateria:"#1a1a05",tierra:"#101010",bobina:"#1a1005",ref5v:"#051a15" };

const CATS = [
  { label:"TODOS",        cat:null           },
  { label:"SENSORES",     cat:"SENSOR"       },
  { label:"ACTUADORES",   cat:"ACTUADOR"     },
  { label:"ALIMENTACIÓN", cat:"ALIMENTACION" },
  { label:"TIERRAS",      cat:"TIERRA"       },
];

function CompBox({ x, y, w=134, h=44, name, tipo, selected, dimmed }) {
  const fill   = dimmed ? "#070a14" : (TIPO_FILL[tipo]  || "#0d1120");
  const stroke = selected ? "#60a5fa" : dimmed ? "#0c1527" : (TIPO_STROKE[tipo] || "#334155");
  const sw     = selected ? 2 : 1.5;
  const tc     = dimmed ? "#1a2a3a" : "#e2e8f0";
  const sc     = dimmed ? "#0f1e2e" : "#64748b";
  const ic     = dimmed ? "#0f1e2e" : (TIPO_STROKE[tipo] || "#64748b");
  return (
    <g>
      {selected && <rect x={x-3} y={y-3} width={w+6} height={h+6} rx={9} fill={TIPO_STROKE[tipo]||"#3b82f6"} opacity={0.15}/>}
      <rect x={x} y={y} width={w} height={h} rx={6} fill={fill} stroke={stroke} strokeWidth={sw}/>
      <circle cx={x} cy={y+h/2} r={4} fill={dimmed?"#0d1527":"#cbd5e1"} stroke="#222" strokeWidth={1}/>
      <text x={x+13} y={y+18} fill={ic} fontSize={14} fontFamily="monospace" textAnchor="middle">
        {TIPO_ICON[tipo]||"□"}
      </text>
      <text x={x+24} y={y+17} fill={tc} fontSize={10} fontFamily="'Courier New',monospace" fontWeight="bold">
        {name.length>15 ? name.slice(0,15)+"…" : name}
      </text>
      <text x={x+24} y={y+31} fill={sc} fontSize={8} fontFamily="monospace">
        {(tipo||"").toUpperCase()}
      </text>
    </g>
  );
}

export default function PulsarPlatina() {
  const [catFilter, setCatFilter] = useState("TODOS");
  const [selPin,    setSelPin]    = useState(null);
  const [hoverPin,  setHoverPin]  = useState(null);

  const filtered = useMemo(() => {
    const cat = CATS.find(c=>c.label===catFilter)?.cat;
    return cat ? PLATINA_PINS.filter(p=>p.cat===cat) : PLATINA_PINS;
  }, [catFilter]);

  const components = useMemo(() => {
    const seen = new Map();
    filtered.forEach(p => {
      if (!seen.has(p.comp)) seen.set(p.comp, { name:p.comp, tipo:TIPO_MAP[p.comp]||"sensor", pins:[] });
      seen.get(p.comp).pins.push(p.num);
    });
    const arr = Array.from(seen.values());
    arr.forEach((c,i) => { c.y = 46 + i*64; });
    return arr;
  },[filtered]);

  const compYC = useMemo(() => {
    const m = new Map();
    components.forEach(c => m.set(c.name, c.y+22));
    return m;
  },[components]);

  const PCM_X=18, PCM_W=172, PIN_H=25, PIN_Y0=60;
  const CMP_X=710, CMP_W=134, CMP_H=44;
  const SVG_W=900;
  const SVG_H=Math.max(360, components.length*64+110);

  function bezier(x1,y1,x2,y2) {
    const mx=(x1+x2)/2;
    return `M ${x1} ${y1} C ${mx} ${y1}, ${mx} ${y2}, ${x2} ${y2}`;
  }

  const activeComp = selPin ? filtered.find(p=>p.num===selPin)?.comp : null;
  const hoverComp  = hoverPin ? filtered.find(p=>p.num===hoverPin)?.comp : null;

  return (
    <div style={{background:"#060911",minHeight:"100vh",padding:16,fontFamily:"'Courier New',monospace",userSelect:"none"}}>

      {/* Header */}
      <div style={{display:"flex",alignItems:"center",gap:12,marginBottom:14,flexWrap:"wrap"}}>
        <div style={{background:"#b91c1c",borderRadius:8,padding:"6px 14px",color:"#fff",fontWeight:"bold",fontSize:14,letterSpacing:1,display:"flex",gap:6,alignItems:"center"}}>
          <span style={{fontSize:16}}>◉</span> PULSAR PREDICTIVO
        </div>
        <div style={{background:"#0d1a0d",border:"1px solid #1a3a1a",borderRadius:6,padding:"4px 10px",color:"#4ade80",fontSize:11}}>
          Nissan Platina · ECM 10 Terminales · {PLATINA_PINS.length} pines
        </div>
        <div style={{marginLeft:"auto",color:"#1e293b",fontSize:10}}>DATESEG MX</div>
      </div>

      {/* Filtros */}
      <div style={{display:"flex",gap:8,marginBottom:14,flexWrap:"wrap"}}>
        {CATS.map(({label,cat})=>{
          const count = cat ? PLATINA_PINS.filter(p=>p.cat===cat).length : PLATINA_PINS.length;
          const active = catFilter===label;
          return (
            <button key={label} onClick={()=>{setCatFilter(label);setSelPin(null);}}
              style={{padding:"5px 14px",borderRadius:20,border:active?"none":"1px solid #1e2535",
                cursor:"pointer",fontSize:11,fontFamily:"monospace",
                background:active?"#b91c1c":"#0d1120",
                color:active?"#fff":"#94a3b8",fontWeight:active?"bold":"normal"}}>
              {label} ({count})
            </button>
          );
        })}
        {selPin && (
          <button onClick={()=>setSelPin(null)}
            style={{padding:"5px 12px",borderRadius:20,border:"1px solid #b91c1c",
              background:"transparent",color:"#f87171",cursor:"pointer",fontSize:11}}>
            ✕ Limpiar
          </button>
        )}
      </div>

      {/* Diagrama SVG */}
      <div style={{overflowX:"auto",overflowY:"auto",background:"#07090f",borderRadius:12,border:"1px solid #0f172a",padding:6}}>
        <svg width={SVG_W} height={SVG_H}>
          <defs>
            <filter id="glow2">
              <feGaussianBlur stdDeviation="3.5" result="blur"/>
              <feMerge><feMergeNode in="blur"/><feMergeNode in="SourceGraphic"/></feMerge>
            </filter>
          </defs>

          {/* PCM sombra */}
          <rect x={PCM_X+3} y={38} width={PCM_W} height={filtered.length*PIN_H+24} rx={10} fill="#000" opacity={0.5}/>
          {/* PCM bloque */}
          <rect x={PCM_X} y={36} width={PCM_W} height={filtered.length*PIN_H+24} rx={10} fill="#08101e" stroke="#1e3a5f" strokeWidth={2}/>
          {/* PCM header */}
          <rect x={PCM_X} y={36} width={PCM_W} height={28} rx={10} fill="#0c1a2e"/>
          <rect x={PCM_X} y={52} width={PCM_W} height={12} fill="#0c1a2e"/>
          <text x={PCM_X+PCM_W/2} y={52} textAnchor="middle" fill="#b91c1c" fontSize={11} fontWeight="bold">ECM / PCM</text>
          <text x={PCM_X+PCM_W/2} y={65} textAnchor="middle" fill="#334155" fontSize={8}>Nissan Platina — 10 Term.</text>

          {/* Cables */}
          {filtered.map((pin,i)=>{
            const py = PIN_Y0 + i*PIN_H + PIN_H/2;
            const cy = compYC.get(pin.comp);
            if (!cy) return null;
            const isSel = selPin===pin.num;
            const isHov = hoverPin===pin.num;
            const dim   = (selPin||hoverPin) && !isSel && !isHov;
            const sw    = isSel?4:isHov?3:dim?0.7:2;
            const op    = dim?0.1:1;
            const d     = bezier(PCM_X+PCM_W, py, CMP_X, cy);
            return (
              <g key={`w${pin.num}`} opacity={op}>
                {isSel && <path d={d} fill="none" stroke={pin.c1} strokeWidth={14} strokeOpacity={0.12} filter="url(#glow2)"/>}
                <path d={d} fill="none" stroke={pin.c1} strokeWidth={sw}/>
                {pin.c2 && (
                  <path d={d} fill="none" stroke={pin.c2}
                    strokeWidth={isSel?2.5:isHov?2:1.2}
                    strokeDasharray={isSel?"11,7":isHov?"8,5":"5,4"}
                    strokeOpacity={0.88}/>
                )}
                {isSel && (
                  <text x={(PCM_X+PCM_W+CMP_X)/2} y={Math.min(py,cy)-10}
                    textAnchor="middle" fill="#fff" fontSize={10} fontFamily="monospace"
                    style={{filter:"drop-shadow(0 0 5px #000)"}}>
                    {pin.cName}
                  </text>
                )}
              </g>
            );
          })}

          {/* Pines en PCM */}
          {filtered.map((pin,i)=>{
            const py    = PIN_Y0 + i*PIN_H + PIN_H/2;
            const isSel = selPin===pin.num;
            const isHov = hoverPin===pin.num;
            const dim   = (selPin||hoverPin) && !isSel && !isHov;
            return (
              <g key={pin.num}
                onClick={()=>setSelPin(isSel?null:pin.num)}
                onMouseEnter={()=>setHoverPin(pin.num)}
                onMouseLeave={()=>setHoverPin(null)}
                style={{cursor:"pointer"}}>
                <rect x={PCM_X+5} y={py-10} width={PCM_W-10} height={20} rx={3}
                  fill={isSel?"#0c1f40":isHov?"#0a1830":dim?"transparent":"#0b1425"}
                  stroke={isSel?"#2563eb":isHov?"#1d4ed8":"transparent"} strokeWidth={1}/>
                {/* Swatch color 1 */}
                <rect x={PCM_X+9} y={py-8} width={13} height={16} rx={2}
                  fill={pin.c1} stroke="#111" strokeWidth={0.5}/>
                {/* Swatch color 2 */}
                {pin.c2 && (
                  <rect x={PCM_X+9} y={py+0} width={13} height={8} rx={0}
                    fill={pin.c2} stroke="#111" strokeWidth={0.5}/>
                )}
                {/* Número pin */}
                <text x={PCM_X+26} y={py+4}
                  fill={dim?"#1a2c3c":isSel?"#93c5fd":isHov?"#60a5fa":"#3b82f6"}
                  fontSize={10} fontWeight="bold">{pin.num}</text>
                {/* Función */}
                <text x={PCM_X+62} y={py+4}
                  fill={dim?"#0f1e2c":isSel?"#94a3b8":"#475569"} fontSize={8}>
                  {pin.fn.length>14 ? pin.fn.slice(0,14)+"…" : pin.fn}
                </text>
                {/* Terminal de salida */}
                <circle cx={PCM_X+PCM_W} cy={py} r={3.5}
                  fill={dim?"#0c1527":pin.c1} stroke="#0a0e1a" strokeWidth={1}/>
              </g>
            );
          })}

          {/* Componentes */}
          {components.map(comp=>{
            const isSel = activeComp===comp.name || hoverComp===comp.name;
            const dim   = (selPin||hoverPin) && !isSel;
            return (
              <CompBox key={comp.name}
                x={CMP_X} y={comp.y} w={CMP_W} h={CMP_H}
                name={comp.name} tipo={comp.tipo}
                selected={isSel} dimmed={dim}/>
            );
          })}

          {/* Badges pines múltiples */}
          {components.map(comp=>{
            const dim = (selPin||hoverPin) && activeComp!==comp.name;
            if (comp.pins.length<2) return null;
            return (
              <text key={`b${comp.name}`}
                x={CMP_X+CMP_W+5} y={comp.y+26}
                fill={dim?"#0c1527":"#2d3a4a"} fontSize={7} fontFamily="monospace">
                {comp.pins.join("·")}
              </text>
            );
          })}

          {/* Líneas guía verticales */}
          <line x1={PCM_X+PCM_W+16} y1={36} x2={PCM_X+PCM_W+16} y2={SVG_H-16} stroke="#0d1527" strokeWidth={1} strokeDasharray="3,5"/>
          <line x1={CMP_X-16} y1={36} x2={CMP_X-16} y2={SVG_H-16} stroke="#0d1527" strokeWidth={1} strokeDasharray="3,5"/>

          {/* Watermark */}
          <text x={SVG_W-10} y={SVG_H-8} textAnchor="end" fill="#0c1527" fontSize={8} fontFamily="monospace">
            DATESEG MX © PULSAR PREDICTIVO v8
          </text>
        </svg>
      </div>

      {/* Panel info pin */}
      {(()=>{
        const pin = selPin ? PLATINA_PINS.find(p=>p.num===selPin) : null;
        if (!pin) return (
          <div style={{marginTop:10,padding:"8px 14px",color:"#1e3a5f",fontSize:10,textAlign:"center"}}>
            ← Toca un pin del ECM para ver su circuito y color de cable
          </div>
        );
        const tipo   = TIPO_MAP[pin.comp]||"sensor";
        const accent = TIPO_STROKE[tipo]||"#334155";
        return (
          <div style={{marginTop:12,background:"#080c18",border:`1.5px solid ${accent}`,borderRadius:10,
            padding:"10px 16px",display:"flex",gap:18,alignItems:"center",flexWrap:"wrap"}}>
            <div style={{display:"flex",gap:4,alignItems:"center",flexShrink:0}}>
              <div style={{width:32,height:32,background:pin.c1,borderRadius:5,border:"1px solid #333"}}/>
              {pin.c2 && <div style={{width:32,height:16,background:pin.c2,borderRadius:3,border:"1px solid #333"}}/>}
            </div>
            <div>
              <div style={{color:"#60a5fa",fontSize:14,fontWeight:"bold"}}>PIN {pin.num}</div>
              <div style={{color:"#e2e8f0",fontSize:11}}>{pin.fn}</div>
            </div>
            <div style={{fontSize:11}}><span style={{color:"#475569"}}>Cable: </span><span style={{color:"#e2e8f0"}}>{pin.cName}</span></div>
