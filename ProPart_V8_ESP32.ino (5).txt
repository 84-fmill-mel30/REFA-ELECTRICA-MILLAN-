/*
 * ProPart V8 — ESP32 Backend
 * Lee módulo OBD2 (Vgate iCar Pro) vía Bluetooth
 * Publica sensores a Firebase en tiempo real
 * Escucha comandos de actuadores desde la app
 * 
 * Hardware:
 *   - ESP32 (DevKit C o similar)
 *   - Vgate iCar Pro (Bluetooth OBD2)
 *   - WiFi de la casa/taller
 * 
 * Setup:
 *   1. Instala bibliotecas en Arduino IDE:
 *      - Firebase Realtime Database (versión 1.2.1 o compatible)
 *      - BluetoothSerial (ya viene con ESP32)
 *   2. Reemplaza WiFi SSID/PASSWORD y Firebase keys
 *   3. Carga en el ESP32
 *   4. Empareja Vgate con la MAC que sale en Serial Monitor
 *   5. Abre el Serial Monitor a 115200 baud
 */

// VARIABLES GLOBALES
// ════════════════════════════════════════════════════════════════════
BluetoothSerial SerialBT;
FirebaseData fbdo;
unsigned long lastPublish = 0;
const unsigned long PUBLISH_INTERVAL = 2000;  // publican cada 2 segundos

// Sensores actuales (para leer y publicar)
struct Sensores {
  float rpm = 0;
  float speed = 0;
  float temp = 85;  // temperatura motor
  float tps = 0;    // posición acelerador
  float maf = 0;    // flujo aire
  float map_ = 0;   // presión múltiple
  float o2 = 0;     // sonda oxígeno
  float volt = 13.5;
  float iat = 25;   // temperatura aire entrada
  float carga = 0;  // carga motor %
  uint32_t mil = 0; // luz Check Engine (0=apagada, 1=encendida)
  uint32_t timestamp = 0;
} sensores;

// DTCs (códigos de falla)
struct DTC {
  String codigo;    // P0101, P0300, etc
  String descripcion;
  String estado;    // "activo" o "inactivo"
};
DTC dtcActual = {"", "", "activo"};

// Actuadores en ejecución
struct Actuador {
  String tipo;      // "bomba_combustible", "ventilador", "test_inyector", etc
  String estado;    // "encendido", "apagado", "pendiente"
  uint32_t duracion = 0;  // ms si es un test
} actuador;

// ════════════════════════════════════════════════════════════════════
// SETUP
// ════════════════════════════════════════════════════════════════════
void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println("\n\n═══════════════════════════════════════════");
  Serial.println("ProPart V8 — ESP32 OBD Backend");
  Serial.println("═══════════════════════════════════════════");
  
  // Bluetooth
  Serial.println("[BLE] Iniciando Bluetooth...");
  SerialBT.begin("ProPart_V8_ESP32");  // nombre del ESP32 visible en búsqueda BLE
  Serial.println("[BLE] ✓ Esperando conexión Vgate");
  Serial.println("[BLE] Empareja el Vgate iCar Pro con: ProPart_V8_ESP32");
  
  // WiFi
  Serial.print("[WiFi] Conectando a "); Serial.println(WIFI_SSID);
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  int intentos = 0;
  while (WiFi.status() != WL_CONNECTED && intentos < 20) {
    delay(500);
    Serial.print(".");
    intentos++;
  }
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\n[WiFi] ✓ Conectado");
    Serial.print("[WiFi] IP: "); Serial.println(WiFi.localIP());
  } else {
    Serial.println("\n[WiFi] ✗ FALLO — revisa SSID y password");
  }
  
  // Firebase
  Serial.println("[Firebase] Inicializando...");
  Firebase.begin(FIREBASE_HOST, FIREBASE_AUTH);
  Firebase.reconnectWiFi(true);
  Firebase.setReadTimeout(fbdo, 5000);
  Firebase.setwriteSizeLimit(fbdo, "small");
  Serial.println("[Firebase] ✓ Listo");
  
  // Iniciar lectura de sensores
  Serial.println("[OBD] Inicializando ELM327...");
  delay(2000);  // esperar a que Vgate se conecte
  obdInit();
  
  Serial.println("═══════════════════════════════════════════");
  Serial.println("Sistema listo. Leyendo sensores en vivo...");
  Serial.println("═══════════════════════════════════════════\n");
}

// ════════════════════════════════════════════════════════════════════
// LOOP
// ════════════════════════════════════════════════════════════════════
void loop() {
  // ── Leer datos del Vgate en tiempo real ──
  if (SerialBT.available()) {
    String response = readVgateResponse();
    parseOBDResponse(response);
  }
  
  // ── Escuchar comandos de la app (actuadores) ──
  if (Firebase.ready()) {
    listenActuadores();
  }
  
  // ── Publicar sensores a Firestore cada 2 seg ──
  if (millis() - lastPublish > PUBLISH_INTERVAL && WiFi.status() == WL_CONNECTED && Firebase.ready()) {
    publishSensores();
    lastPublish = millis();
  }
  
  // ── Escaneo de DTCs cada 30 seg ──
  static unsigned long lastDTC = 0;
  if (millis() - lastDTC > 30000) {
    scanDTC();
    lastDTC = millis();
  }
  
  delay(50);
}

// ════════════════════════════════════════════════════════════════════
// OBD2 — INICIALIZACIÓN
// ════════════════════════════════════════════════════════════════════
void obdInit() {
  SerialBT.println("ATZ");      // reset
  delay(1000);
  flushBT();
  
  SerialBT.println("ATE0");     // echo off
  delay(500);
  flushBT();
  
  SerialBT.println("ATL0");     // linefeed off
  delay(500);
  flushBT();
  
  SerialBT.println("ATS0");     // espacios off
  delay(500);
  flushBT();
  
  SerialBT.println("ATSP0");    // protocolo automático
  delay(500);
  flushBT();
  
  Serial.println("[OBD] ✓ ELM327 inicializado");
  
  // test: pedir RPM (PID 010C)
  delay(1000);
  obdRequest("010C");
}

// ════════════════════════════════════════════════════════════════════
// OBD2 — PETICIÓN
// ════════════════════════════════════════════════════════════════════
void obdRequest(String pid) {
  SerialBT.println(pid);
  delay(150);
}

// ════════════════════════════════════════════════════════════════════
// OBD2 — LECTURA DE RESPUESTA
// ════════════════════════════════════════════════════════════════════
String readVgateResponse() {
  String response = "";
  unsigned long timeout = millis() + 2000;
  while (millis() < timeout && SerialBT.available()) {
    char c = SerialBT.read();
    if (c == '\r' || c == '\n') {
      if (response.length() > 2) break;
    } else {
      response += c;
    }
    delay(10);
  }
  return response;
}

// ════════════════════════════════════════════════════════════════════
// OBD2 — PARSEAR RESPUESTAS
// ════════════════════════════════════════════════════════════════════
void parseOBDResponse(String resp) {
  if (resp.length() < 4) return;
  
  resp.toUpperCase();
  resp.trim();
  
  // Si el Vgate devuelve NO DATA o error, pasar
  if (resp.indexOf("NO DATA") >= 0 || resp.indexOf("?" ) >= 0) {
    Serial.println("[OBD] ⚠️ ECU no responde — ¿llave en ON?");
    return;
  }
  
  // ─── RPM: PID 010C → respuesta 41 0C XX XX ───
  if (resp.startsWith("410C")) {
    int a = hexToDec(resp.substring(4, 6));
    int b = hexToDec(resp.substring(6, 8));
    sensores.rpm = (a * 256 + b) / 4.0;
    Serial.print("[OBD] RPM: "); Serial.println(sensores.rpm);
  }
  // ─── Velocidad: 010D → 41 0D XX ───
  else if (resp.startsWith("410D")) {
    sensores.speed = hexToDec(resp.substring(4, 6));
    Serial.print("[OBD] Speed: "); Serial.println(sensores.speed);
  }
  // ─── Temperatura motor: 0105 → 41 05 XX ───
  else if (resp.startsWith("4105")) {
    sensores.temp = hexToDec(resp.substring(4, 6)) - 40;
    Serial.print("[OBD] Temp: "); Serial.println(sensores.temp);
  }
  // ─── TPS: 0111 → 41 11 XX ───
  else if (resp.startsWith("4111")) {
    sensores.tps = hexToDec(resp.substring(4, 6)) * 100.0 / 255;
    Serial.print("[OBD] TPS: "); Serial.println(sensores.tps);
  }
  // ─── MAF: 0110 → 41 10 XX XX ───
  else if (resp.startsWith("4110")) {
    int a = hexToDec(resp.substring(4, 6));
    int b = hexToDec(resp.substring(6, 8));
    sensores.maf = (a * 256 + b) / 100.0;
    Serial.print("[OBD] MAF: "); Serial.println(sensores.maf);
  }
  // ─── MAP: 010B → 41 0B XX ───
  else if (resp.startsWith("410B")) {
    sensores.map_ = hexToDec(resp.substring(4, 6));
    Serial.print("[OBD] MAP: "); Serial.println(sensores.map_);
  }
  // ─── Sonda O2: 0114 → 41 14 XX XX ───
  else if (resp.startsWith("4114")) {
    int a = hexToDec(resp.substring(4, 6));
    sensores.o2 = a / 200.0;
    Serial.print("[OBD] O2: "); Serial.println(sensores.o2);
  }
  // ─── Voltaje: 0142 → 41 42 XX XX ───
  else if (resp.startsWith("4142")) {
    int a = hexToDec(resp.substring(4, 6));
    int b = hexToDec(resp.substring(6, 8));
    sensores.volt = (a * 256 + b) / 1000.0;
    Serial.print("[OBD] Volt: "); Serial.println(sensores.volt);
  }
  // ─── DTCs modo 03: respuesta 43 XX XX ... ───
  else if (resp.startsWith("43")) {
    parseDTC_Modo03(resp);
  }
  // ─── DTCs modo 07 (pendientes): 47 XX XX ... ───
  else if (resp.startsWith("47")) {
    Serial.println("[OBD] DTCs pendientes encontrados");
  }
  
  sensores.timestamp = millis() / 1000;
}

void parseDTC_Modo03(String resp) {
  // Formato: 43 [conteo] XX XX (código de 2 bytes)
  // Ejemplo: 43 01 P0101 → 43 P 01 01
  if (resp.length() < 6) return;
  
  // Extraer el primer DTC (primeros 4 hex tras "43")
  String hex = resp.substring(2, 6);
  dtcActual.codigo = decodeDTC(hex);
  dtcActual.estado = "activo";
  
  // Buscar descripción
  if (dtcActual.codigo == "P0101") dtcActual.descripcion = "MAF sensor flow";
  else if (dtcActual.codigo == "P0300") dtcActual.descripcion = "Multiple/random misfire";
  else if (dtcActual.codigo == "P0171") dtcActual.descripcion = "Lean fuel trim bank 1";
  else if (dtcActual.codigo == "P0172") dtcActual.descripcion = "Rich fuel trim bank 1";
  else dtcActual.descripcion = "Check manual";
  
  Serial.print("[OBD] DTC encontrado: "); Serial.println(dtcActual.codigo);
}

String decodeDTC(String hex) {
  if (hex.length() < 4) return "XXXX";
  int b1 = hexToDec(hex.substring(0, 2));
  char letras[] = {'P', 'C', 'B', 'U'};
  char letra = letras[(b1 >> 6) & 0x03];
  int d1 = (b1 >> 4) & 0x03;
  int d2 = b1 & 0x0F;
  String d34 = hex.substring(2, 4);
  String code = "";
  code += letra;
  code += d1;
  code += String(d2, HEX);
  code += d34;
  code.toUpperCase();
  return code;
}

// ════════════════════════════════════════════════════════════════════
// OBD2 — SCAN DTCs
// ════════════════════════════════════════════════════════════════════
void scanDTC() {
  if (!SerialBT.connected()) return;
  Serial.println("[OBD] Escaneando DTCs (modo 03)...");
  obdRequest("03");
  delay(800);
  String response = readVgateResponse();
  parseOBDResponse(response);
}

// ════════════════════════════════════════════════════════════════════
// FIREBASE — PUBLICAR SENSORES
// ════════════════════════════════════════════════════════════════════
void publishSensores() {
  FirebaseJson json;
  
  // Sensores como mapa (estructura correcta)
  json.set("sensores/rpm", sensores.rpm);
  json.set("sensores/speed", sensores.speed);
  json.set("sensores/temp", sensores.temp);
  json.set("sensores/tps", sensores.tps);
  json.set("sensores/maf", sensores.maf);
  json.set("sensores/map", sensores.map_);
  json.set("sensores/o2", sensores.o2);
  json.set("sensores/volt", sensores.volt);
  json.set("sensores/iat", sensores.iat);
  json.set("sensores/carga", sensores.carga);
  json.set("sensores/mil", sensores.mil);
  json.set("sensores/timestamp", sensores.timestamp);
  
  // DTCs
  if (dtcActual.codigo.length() > 0) {
    json.set("dtcs/codigo", dtcActual.codigo);
    json.set("dtcs/descripcion", dtcActual.descripcion);
    json.set("dtcs/estado", dtcActual.estado);
  }
  
  // Config
  json.set("config/dispositivo", "Vgate iCar Pro");
  json.set("config/direccion", "Bluetooth");
  json.set("config/SENSORES_ACTIVOS", 10);
  
  // Actualizar documento "default" con merge
  if (Firebase.updateNode(fbdo, "/propartv8Bidireccional/default", json)) {
    Serial.print("[Firebase] ✓ Publicado | RPM: ");
    Serial.print(sensores.rpm); Serial.print(" | Temp: ");
    Serial.println(sensores.temp);
  } else {
    Serial.print("[Firebase] ✗ Error: ");
    Serial.println(fbdo.errorReason());
  }
}

// ════════════════════════════════════════════════════════════════════
// FIREBASE — ESCUCHAR COMANDOS DE ACTUADORES
// ════════════════════════════════════════════════════════════════════
void listenActuadores() {
  static unsigned long lastCheck = 0;
  if (millis() - lastCheck < 1000) return;  // chequear cada 1 seg
  lastCheck = millis();
  
  if (Firebase.getJSON(fbdo, "/propartv8Bidireccional/default/actuadores/comando")) {
    FirebaseJson &json = fbdo.jsonObject();
    String tipo = json.stringValue();
    
    if (tipo.length() > 0 && tipo != "null") {
      Serial.print("[Actuadores] Comando recibido: ");
      Serial.println(tipo);
      
      // ─── Ejecutar comandos según el tipo ───
      if (tipo.indexOf("INJ") >= 0) {
        // Test inyector: INJ1:75, INJ2:75, etc
        Serial.println("[Act] Test inyector (simulado)");
      }
      else if (tipo.indexOf("EGR") >= 0) {
        // Control EGR
        Serial.println("[Act] EGR ajustado");
      }
      else if (tipo.indexOf("bomba") >= 0) {
        // Bomba de combustible
        Serial.println("[Act] Bomba de combustible ON");
      }
      else if (tipo.indexOf("ventilador") >= 0) {
        // Ventilador
        Serial.println("[Act] Ventilador ON");
      }
      
      // Responder a la app
      respondActuador(tipo, "ok", "Ejecutado correctamente");
    }
  }
}

void respondActuador(String tipo, String estado, String mensaje) {
  FirebaseJson json;
  json.set("respuesta/tipo", tipo);
  json.set("respuesta/estado", estado);
  json.set("respuesta/mensaje", mensaje);
  json.set("respuesta/ts", (int)(millis() / 1000));
  
  Firebase.updateNode(fbdo, "/propartv8Bidireccional/default/actuadores", json);
}

// ════════════════════════════════════════════════════════════════════
// UTILIDADES
// ════════════════════════════════════════════════════════════════════
int hexToDec(String hex) {
  return strtol(hex.c_str(), NULL, 16);
}

void flushBT() {
  while (SerialBT.available()) SerialBT.read();
}
