function CompBox({ x, y, w=140, h=44, name, tipo, selected, dimmed }) {
  const fill   = dimmed ? "#070a14" : (TIPO_FILL[tipo]  || "#0d1120");
  const stroke = selected ? "#60a5fa" : dimmed ? "#0d1527" : (TIPO_STROKE[tipo] || "#334155");
  const sw     = selected ? 2 : 1.5;
  const tc     = dimmed ? "#1a2a3a" : "#e2e8f0";
  const sc     = dimmed ? "#0f1e2e" : "#64748b";
  const ic     = dimmed ? "#0f1e2e" : (TIPO_STROKE[tipo] || "#64748b");

  return (
    <g>
      {selected && <rect x={x-3} y={y-3} width={w+6} height={h+6} rx={9} fill={TIPO_STROKE[tipo]||"#3b82f6"} opacity={0.12}/>}
      <rect x={x} y={y} width={w} height={h} rx={6} fill={fill} stroke={stroke} strokeWidth={sw}/>
      <circle cx={x} cy={y+h/2} r={4} fill={dimmed?"#0d1527":"#cbd5e1"} stroke="#222" strokeWidth={1}/>
      <text x={x+12} y={y+18} fill={ic} fontSize={14} fontFamily="monospace" textAnchor="middle">
        {TIPO_ICON[tipo]||"□"}
      </text>
      <text x={x+22} y={y+17} fill={tc} fontSize={9} fontFamily="'Courier New',monospace" fontWeight="bold">
        {name.length>16 ? name.slice(0,16)+"…" : name}
      </text>
      <text x={x+22} y={y+31} fill={sc} fontSize={8} fontFamily="monospace">
        {(tipo||"").toUpperCase()}
      </text>
    </g>
  );
}

      const q = search.toLowerCase();

        p.num.toLowerCase().includes(q) ||

                </text>
                {/* Función */}
                <text x={PCM_X+58} y={py+17} fill={isSelected?"#e2e8f0":"#94a3b8"}
                  fontSize={9} fontFamily="monospace">
                  {pin.fn.length>22 ? pin.fn.slice(0,22)+"…" : pin.fn}
                </text>
                {/* Muestra de color */}
                <rect x={PCM_X+PCM_W-24} y={py+6} width={10} height={10} rx={2}
                  fill={pin.c1} stroke="#222" strokeWidth={0.5}/>
                {pin.c2 && <rect x={PCM_X+PCM_W-13} y={py+6} width={10} height={10} rx={2}
                  fill={pin.c2} stroke="#222" strokeWidth={0.5}/>}
                {/* Punto conector */}
                <circle cx={PCM_X+PCM_W} cy={py+PIN_H/2} r={3}
                  fill={isSelected?"#60a5fa":"#1e3a5f"}/>
              </g>
            );
          })}

          {/* Componentes lado derecho */}
          {components.map(c=>{
            const isActive = !activeComp || activeComp===c.name;
            const isSelected = activeComp===c.name;
            return (
              <CompBox key={c.name}
                x={CMP_X} y={c.y} w={CMP_W} h={CMP_H}
                name={c.name} tipo={c.tipo}
                selected={isSelected} dimmed={!isActive}/>
            );
          })}
        </svg>
      </div>

      {/* Leyenda */}
      <div style={{marginTop:14,padding:"10px 14px",background:"#0d1120",borderRadius:8,
        border:"1px solid #1e2535",display:"flex",gap:16,flexWrap:"wrap"}}>
        {Object.entries(TIPO_ICON).map(([tipo,icon])=>(
          <div key={tipo} style={{display:"flex",alignItems:"center",gap:5}}>
            <span style={{color:TIPO_STROKE[tipo]||"#64748b",fontSize:14}}>{icon}</span>
            <span style={{color:"#475569",fontSize:10}}>{tipo.toUpperCase()}</span>
          </div>
        ))}
        <div style={{marginLeft:"auto",color:"#1e293b",fontSize:9}}>
          {GOLF_PINS.length} pines · Ed.09.03/04.04
        </div>
      </div>
    </div>
  );
}
